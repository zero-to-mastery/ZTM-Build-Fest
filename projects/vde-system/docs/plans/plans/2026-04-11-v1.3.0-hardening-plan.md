```zsh
#!/usr/bin/env zsh
# VDE Pre-Commit Sentinel - Rule C & Rule 12 Enforcement

# 1. Zsh Purity Check (Rule C)
for file in $(git diff --cached --name-only --filter=ACM | grep -E '\.(zsh|sh)$'); do
    if head -n 1 "$file" | grep -qE '#!/bin/(bash|sh)'; then
        echo "[CRITICAL] Rule C Violation: Non-Zsh shebang found in $file" >&2
        exit 1
    fi
done

# 2. Secret Sentinel (Rule 12)
if git diff --cached | grep -Ei "API_KEY|SECRET|PASSWORD" | grep -v "env.template"; then
    echo "[CRITICAL] Rule 12 Violation: Potential hardcoded secret detected in staged changes." >&2
    exit 1
fi

# 3. Silent Spine Check
./bin/vde-spine-check.zsh >/dev/null || { echo "[CRITICAL] System Spine Failure: Commit blocked." >&2; exit 1; }
```

```zsh
source ./lib/vm-common

echo "# VDE Sovereign Vision Dashboard"
echo "## Active Spokes"
printf "| Spoke | Status | Uptime | Port |\n"
printf "| :--- | :--- | :--- | :--- |\n"

    uptime=$(docker inspect "$container" --format '{{.State.StartedAt}}')
    port=$(docker port "$container" 22 | cut -d: -f2)
    printf "| %s | UP | %s | %s |\n" "$container" "$uptime" "$port"
done
```
